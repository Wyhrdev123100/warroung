    const out = { ...a };
    for (const k of Object.keys(b || {})) {
      if (b[k] && typeof b[k] === "object" && !Array.isArray(b[k]))
        out[k] = deepMerge(a[k] || {}, b[k]);
      else out[k] = b[k];
    }
    return out;

  async function httpRequest({
    url,
    method = "GET",
    headers = {},
    data = null,
    timeout = 45000,
  }) {
    if (typeof GM_xmlhttpRequest === "function") {
      return new Promise((resolve, reject) => {
        GM_xmlhttpRequest({
          method,
          url,
          headers,
          data,
          timeout,
          onload: (res) =>
            res.status >= 200 && res.status < 300
              ? resolve(res.responseText)
              : reject(
                  Object.assign(new Error(`HTTP ${res.status}`), {
                    status: res.status,
                    body: res.responseText,
                  }),
                ),
          onerror: () => reject(new Error("Network error")),
          ontimeout: () => reject(new Error("Timeout")),
        });
      });
    }
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeout);
    try {
      const res = await fetch(url, {
        method,
        headers,
        body: data,
        signal: controller.signal,
        credentials: "omit",
        mode: "cors",
      });
      const text = await res.text();
      if (!res.ok)
        throw Object.assign(new Error(`HTTP ${res.status}`), {
          status: res.status,
          body: text,
        });
      return text;
    } finally {
      clearTimeout(timer);
    }
  }

  function parseJsonLoose(raw) {
    const cleaned = String(raw)
      .replace(/```(?:json)?/gi, "")
      .replace(/```/g, "")
      .trim();
    const start = cleaned.indexOf("{");
    const end = cleaned.lastIndexOf("}");
    if (start === -1 || end === -1) throw new Error("AI không trả JSON");
    return JSON.parse(cleaned.slice(start, end + 1));
  }

  function resolveTarget(answer, options) {
    if (!answer) return null;
    const byText = options.find((o) => {
      const t = normalize(o.text),
        a = normalize(answer.answer_text);
      return (
        a &&
        (t === a ||
          (a.length > 6 && t.includes(a)) ||
          (t.length > 6 && a.includes(t)))
      );
    });
    if (byText) return byText;
    const idx = Number(answer.answer_index);
    if (Number.isInteger(idx) && options[idx - 1]) return options[idx - 1];
    const head = normalize(answer.answer_text).slice(0, 25);
    return options.find((o) => normalize(o.text).startsWith(head)) || null;
  }

  function vote(answers, options) {
    const tally = new Map();
    answers.forEach(({ provider, answer }) => {
      const target = resolveTarget(answer, options);
      if (!target) return;
      const e = tally.get(target.index) || {
        index: target.index,
        text: target.text,
        votes: 0,
        providers: [],
        confSum: 0,
      };
      e.votes++;
      e.providers.push(provider);
      e.confSum += Number(answer.confidence) || 0.5;
      tally.set(target.index, e);
    });
    return [...tally.values()].sort((a, b) =>
      b.votes !== a.votes
        ? b.votes - a.votes
        : b.confSum / b.votes - a.confSum / a.votes,
    );
  }

      autoRun: panel.querySelector('[data-act="autoRun"]'),
      autoClick: panel.querySelector('[data-act="autoClick"]'),
    };

    // restore pos
    if (state.settings.panelPos) {
      panel.style.left = state.settings.panelPos.left + "px";
      panel.style.top = state.settings.panelPos.top + "px";
      panel.style.right = "auto";
    }

    // drag
    const head = panel.querySelector(".wg-head");
    let dragging = false,
      sx = 0,
      sy = 0,
      ox = 0,
      oy = 0;
    head.addEventListener("mousedown", (e) => {
      if (e.target.closest("button")) return;
      dragging = true;
      const r = panel.getBoundingClientRect();
      sx = e.clientX;
      sy = e.clientY;
      ox = r.left;
      oy = r.top;
      panel.style.left = ox + "px";
      panel.style.top = oy + "px";
      panel.style.right = "auto";
      e.preventDefault();
    });
    window.addEventListener("mousemove", (e) => {
      if (!dragging) return;
      const nx = Math.max(
        0,
        Math.min(window.innerWidth - 100, ox + e.clientX - sx),
      );
      const ny = Math.max(
        0,
        Math.min(window.innerHeight - 60, oy + e.clientY - sy),
      );
      panel.style.left = nx + "px";
      panel.style.top = ny + "px";
    });
    window.addEventListener("mouseup", () => {
      if (!dragging) return;
      dragging = false;
      const r = panel.getBoundingClientRect();
      state.settings.panelPos = { left: r.left, top: r.top };
      saveSettings();
    });

    updateToggles();

    panel.addEventListener("click", (e) => {
      const row = e.target.closest(".wg-opt");
      if (row) return chooseOption(Number(row.dataset.index));
      const btn = e.target.closest("button");
      if (!btn) return;
      const act = btn.dataset?.act;
      if (act === "run") solve({ force: true });
      if (act === "cfg") openSettingsModal();
      if (act === "providers") openProvidersModal();
      if (act === "toggle") {
        const c = panel.classList.toggle("collapsed");
        btn.textContent = c ? "+" : "—";

      if (act === "autoRun") {
        state.settings.autoRun = !state.settings.autoRun;
        saveSettings();
        updateToggles();

      }
      if (act === "autoClick") {
        state.settings.autoClick = !state.settings.autoClick;
        saveSettings();
        updateToggles();
      }
    });

        e.preventDefault();
        solve({ force: true });
      }
    });

    ensurePanel();
    ui.question.textContent = data.question || "";
    ui.options.innerHTML = "";
    const target = result?.target || null;
    const tally = result?.tally || [];
    const maxVotes = tally[0]?.votes || 0;

    data.options.forEach((opt) => {
      const row = document.createElement("button");
      row.className = "wg-opt";
      row.dataset.index = String(opt.index);
      if (target && opt.index === target.index) row.classList.add("correct");
      else if (target) row.classList.add("dim");
      const votes = tally.find((t) => t.index === opt.index)?.votes || 0;
      row.innerHTML = `<span class="wg-n">${opt.index}</span><span class="wg-t"></span>${votes ? `<span class="wg-votes">${votes}/${maxVotes}</span>` : ""}`;
      row.querySelector(".wg-t").textContent = opt.text;
      ui.options.appendChild(row);
    });

    if (!renderStatusChips() && result && target) {
      const conf = result.avgConfidence
        ? ` · ${Math.round(result.avgConfidence * 100)}%`
        : "";
      ui.status.classList.remove("err");
      ui.status.innerHTML = `<span class="wg-badge">✓ ${target.index}. ${escapeHtml(target.text)}</span>${conf} · ${tally[0]?.votes || 0} phiếu`;
    }

    if (result && target) {
      const reasonText = state.settings.showReason ? result.reason || "" : "";
      ui.reason.textContent = reasonText;
      ui.reason.style.display = reasonText ? "block" : "none";
    } else {
      ui.reason.style.display = "none";
    }

  function readQuestion() {
    const qEl =
      document.querySelector("#questionText .content-slot span") ||
      document.querySelector("#questionText");
    const question = qEl ? qEl.innerText.trim() : "";
    const els = document.querySelectorAll(
      '.options-grid button[role="option"]',
    );
    const options = Array.from(els).map((el, i) => {
      const tEl =
        el.querySelector("#optionText .content-slot span") ||
        el.querySelector("#optionText");
      return {
        index: i + 1,
        el,
        text: (tEl ? tEl.innerText : el.innerText).trim(),
      };
    });
    return { question, options };
  }

  const signatureOf = ({ question, options }) =>
    question + "|" + options.map((o) => normalize(o.text)).join("|");

  function chooseOption(index) {
    const data = readQuestion();
    const opt = data.options.find((o) => o.index === index);
    if (!opt) return;
    opt.el.scrollIntoView({ behavior: "smooth", block: "center" });
    try {
      opt.el.focus({ preventScroll: true });
    } catch (_) {
      opt.el.focus();
    }
    ["pointerdown", "mousedown", "pointerup", "mouseup"].forEach((t) =>
      opt.el.dispatchEvent(new MouseEvent(t, { bubbles: true })),
    );
    opt.el.click();
  }

  // ================= SOLVE =================
  async function solve({ force = false } = {}) {
    const data = readQuestion();
    if (!data.question || data.options.length < 2) {
      setStatus("Chưa thấy câu hỏi/đáp án trên trang.", true);
      return;
    }
    const sig = signatureOf(data);
    state.lastSignature = sig;

    const cached = state.cache[sig];
    if (!force && cached && Date.now() - cached.at < CACHE_TTL) {
      const target = resolveTarget(cached.best, data.options);
      state.result = {
        target,
        tally: cached.tally || [],
        reason: cached.reason,
        avgConfidence: cached.avgConfidence,
      };
      providerStatus = {};
      render(data, state.result);
      if (state.settings.autoClick && target) chooseOption(target.index);
      return;
    }

    if (state.running) return;
    state.running = true;
    const token = ++state.runToken;

    // fallback: luôn dùng groq kể cả khi chưa bật? không — respect setting
    const active = Object.keys(PROVIDERS).filter(
      (id) => state.settings.enabled[id] && state.settings.keys[id],
    );
    if (!active.length) {
      setStatus(
        "Chưa bật provider nào có API key. Mở Providers để cấu hình.",
        true,
      );
      state.running = false;
      return;
    }

    providerStatus = {};
    active.forEach((id) => (providerStatus[id] = "pending"));
    render(data, null);

    const answers = [];
    const queue = [...active];
    const limit = Math.min(state.settings.maxParallel, queue.length);

    async function worker() {
      while (queue.length) {
        const id = queue.shift();
        if (token !== state.runToken) return;
        try {
          const ans = await callProvider(id, data);
          if (token !== state.runToken) return;
          answers.push({ provider: id, answer: ans });
          providerStatus[id] = "ok";
        } catch (e) {
          if (token !== state.runToken) return;
          providerStatus[id] = "fail";
          console.warn("[MultiAI]", id, e.message);
        }
        render(data, null);
      }
    }
    await Promise.all(Array.from({ length: limit }, worker));
    if (token !== state.runToken) {
      state.running = false;
      return;
    }

    if (!answers.length) {
      setStatus("Không provider nào trả lời được. Kiểm tra key/model.", true);
      state.running = false;
      return;
    }

    const tally = vote(answers, data.options);
    if (!tally.length) {
      setStatus("Không có đáp án nào khớp danh sách.", true);
      state.running = false;
      return;
    }

    let winner;
    if (state.settings.consensusMode === "confidence") {
      const confMap = new Map();
      answers.forEach(({ answer }) => {
        const t = resolveTarget(answer, data.options);
        if (!t) return;
        const c = Number(answer.confidence) || 0.5;
        const cur = confMap.get(t.index) || {
          index: t.index,
          text: t.text,
          conf: 0,
          votes: 0,
        };
        cur.conf += c;
        cur.votes += 1;
        confMap.set(t.index, cur);
      });
      winner = [...confMap.values()].sort((a, b) => b.conf - a.conf)[0];
      winner = { ...winner, avgConf: winner.conf / winner.votes };
    } else {
      const top = tally[0];
      const confSums = answers
        .filter(
          ({ answer }) =>
            resolveTarget(answer, data.options)?.index === top.index,
        )
        .map(({ answer }) => Number(answer.confidence) || 0.5);
      const avgConf = confSums.length
        ? confSums.reduce((a, b) => a + b, 0) / confSums.length
        : 0.5;
      winner = { ...top, avgConf };
    }

    const reasonParts = [];
    answers.forEach(({ provider, answer }) => {
      const t = resolveTarget(answer, data.options);
      if (t && t.index === winner.index && answer.reason)
        reasonParts.push(`[${PROVIDERS[provider].label}] ${answer.reason}`);
    });
    const reason = reasonParts.slice(0, 3).join("\n");

    const target = data.options.find((o) => o.index === winner.index);
    state.result = { target, tally, reason, avgConfidence: winner.avgConf };

    state.cache[sig] = {
      at: Date.now(),
      best: {
        answer_index: winner.index,
        answer_text: winner.text,
        confidence: winner.avgConf,
        reason,
      },
      tally,
      reason,
      avgConfidence: winner.avgConf,
    };
    saveCache();
    providerStatus = {};
    render(data, state.result);

    if (state.settings.autoClick && target)
      setTimeout(() => chooseOption(target.index), 300);
    state.running = false;
  }

      setStatus("Đã lưu providers.");
      close();
      solve({ force: true });
    };
  }

  // ================= MENUS + WATCH =================
  function registerMenus() {
    if (typeof GM_registerMenuCommand !== "function") return;
    GM_registerMenuCommand("⚙ Cấu hình", openSettingsModal);
    GM_registerMenuCommand("🔌 Providers", openProvidersModal);
    GM_registerMenuCommand("⟳ Phân tích lại", () => solve({ force: true }));
    GM_registerMenuCommand("👆 Bật/tắt tự chọn", () => {
      state.settings.autoClick = !state.settings.autoClick;
      saveSettings();
      updateToggles();
      setStatus(`Tự chọn: ${state.settings.autoClick ? "BẬT" : "TẮT"}`);
    });
    GM_registerMenuCommand("🗑 Xoá cache", () => {
      state.cache = {};
      saveCache();
      setStatus("Đã xoá cache.");
    });
  }

  function watch() {
    let debounce = null;
    const check = () => {
      clearTimeout(debounce);
      debounce = setTimeout(() => {
        const data = readQuestion();
        const sig = signatureOf(data);
        if (sig === state.lastSignature) return;
        state.lastSignature = sig;
        state.result = null;
        if (data.question && data.options.length >= 2 && state.settings.autoRun)
          solve();
      }, 700);
    };
    new MutationObserver(check).observe(document.body, {
      childList: true,
      subtree: true,
    });
    check();
  }

  registerMenus();
  ensurePanel();
  watch();
})();
